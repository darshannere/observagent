```javascript
const stmtCostByModel = db.prepare(`
  SELECT
    model,
    SUM(total_cost_usd) AS cost,
    COUNT(DISTINCT session_id) AS sessions
  FROM session_cost
  WHERE agent_id = ''
  GROUP BY model
`);
```

```javascript
    sessions,
    todayTotal: todayRow.total,
    models,
  });
});
```
